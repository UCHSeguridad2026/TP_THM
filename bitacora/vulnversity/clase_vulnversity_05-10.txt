# un navegador corriendo también dentro de WSL (firefox-esr vía WSLg) en vez
# del navegador nativo de Windows, que no tiene ruta a la red de la VPN.
$ sudo apt install firefox-esr -y
$ firefox-esr &

# Prueba manual con archivos de extensión común:
$ echo "test" > /tmp/test.txt
$ echo "test" > /tmp/test.php
# Resultado: .txt sube sin problema; .php responde "not allowed" (bloqueado).

# Fuzzing de extensiones con Burp Intruder (ataque Sniper) sobre el campo
